ragel(${name} src/parsers/thttp_parser_url.c ./ragel/thttp_parser_url.rl)
ragel(${name} src/parsers/thttp_parser_message.c ./ragel/thttp_parser_message.rl)
ragel(${name} src/parsers/thttp_parser_header.c ./ragel/thttp_parser_header.rl)
ragel(${name} src/headers/thttp_header_Authorization.c ./ragel/thttp_parser_header_Authorization.rl)
ragel(${name} src/headers/thttp_header_Content_Length.c ./ragel/thttp_parser_header_Content_Length.rl)
ragel(${name} src/headers/thttp_header_Content_Type.c ./ragel/thttp_parser_header_Content_Type.rl)
ragel(${name} src/headers/thttp_header_Dummy.c ./ragel/thttp_parser_header_Dummy.rl)
ragel(${name} src/headers/thttp_header_ETag.c ./ragel/thttp_parser_header_ETag.rl)
ragel(${name} src/headers/thttp_header_Sec_WebSocket_Accept.c ./ragel/thttp_parser_header_Sec_WebSocket_Accept.rl)
ragel(${name} src/headers/thttp_header_Sec_WebSocket_Key.c ./ragel/thttp_parser_header_Sec_WebSocket_Key.rl)
ragel(${name} src/headers/thttp_header_Sec_WebSocket_Protocol.c ./ragel/thttp_parser_header_Sec_WebSocket_Protocol.rl)
