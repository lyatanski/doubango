cmake_minimum_required(VERSION 3.16)
get_filename_component(name ${CMAKE_CURRENT_SOURCE_DIR} NAME)
project(${name})

make_directory(${CMAKE_CURRENT_SOURCE_DIR}/src/parsers)
file(GLOB src src/*.c src/*/*.c)
ragel(${name} src/parsers/tsdp_parser_message.c ./ragel/tsdp_parser_message.rl)
ragel(${name} src/headers/tsdp_header_A.c ./ragel/tsdp_parser_header_A.rl)
ragel(${name} src/headers/tsdp_header_B.c ./ragel/tsdp_parser_header_B.rl)
ragel(${name} src/headers/tsdp_header_C.c ./ragel/tsdp_parser_header_C.rl)
ragel(${name} src/headers/tsdp_header_Dummy.c ./ragel/tsdp_parser_header_Dummy.rl)
ragel(${name} src/headers/tsdp_header_E.c ./ragel/tsdp_parser_header_E.rl)
ragel(${name} src/headers/tsdp_header_I.c ./ragel/tsdp_parser_header_I.rl)
ragel(${name} src/headers/tsdp_header_K.c ./ragel/tsdp_parser_header_K.rl)
ragel(${name} src/headers/tsdp_header_M.c ./ragel/tsdp_parser_header_M.rl)
ragel(${name} src/headers/tsdp_header_O.c ./ragel/tsdp_parser_header_O.rl)
ragel(${name} src/headers/tsdp_header_P.c ./ragel/tsdp_parser_header_P.rl)
ragel(${name} src/headers/tsdp_header_R.c ./ragel/tsdp_parser_header_R.rl)
ragel(${name} src/headers/tsdp_header_S.c ./ragel/tsdp_parser_header_S.rl)
ragel(${name} src/headers/tsdp_header_T.c ./ragel/tsdp_parser_header_T.rl)
ragel(${name} src/headers/tsdp_header_U.c ./ragel/tsdp_parser_header_U.rl)
ragel(${name} src/headers/tsdp_header_V.c ./ragel/tsdp_parser_header_V.rl)
ragel(${name} src/headers/tsdp_header_Z.c ./ragel/tsdp_parser_header_Z.rl)

add_library(${name} ${src} ${RAGEL_${name}_FILES})
target_include_directories(${name} PUBLIC include)
target_link_libraries(${name} tinySAK)

file(GLOB tests test/*.c)
foreach(test_src ${tests})
    get_filename_component(test ${test_src} NAME_WE)
    add_executable(${test}-${name} ${test_src})
    target_link_libraries(${test}-${name} ${name})
    if(APPLE)
        target_link_libraries(${test}-${name} "-framework Foundation" "-framework IOKit")
    endif()
    add_test(${name} ./${test}-${name})
endforeach()
